function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);

    var timestamp = new Date();
    var name = data.name || "";
    var phone = data.phone || "";
    var email = data.email || "";
    var message = data.message || "";
    var source = data.source || "Portfolio Website";

    sheet.appendRow([timestamp, name, phone, email, message, source]);

    MailApp.sendEmail({
      to: "venkatasateeshdivi@gmail.com",
      subject: "New Portfolio Lead - " + name,
      htmlBody:
        "<h2>New Portfolio Lead</h2>" +
        "<p><b>Name:</b> " + name + "</p>" +
        "<p><b>Phone:</b> " + phone + "</p>" +
        "<p><b>Email:</b> " + email + "</p>" +
        "<p><b>Message:</b> " + message + "</p>" +
        "<p><b>Source:</b> " + source + "</p>"
    });

    if (email) {
      MailApp.sendEmail({
        to: email,
        subject: "Thank you for contacting Divi Venkata Sateesh Babu",
        htmlBody:
          "<p>Hi " + name + ",</p>" +
          "<p>Thank you for visiting my portfolio and submitting your details.</p>" +
          "<p>I have received your message and will get back to you soon.</p>" +
          "<br><p>Regards,<br><b>DIVI VENKATA SATEESH BABU</b><br>Performance Marketing Professional</p>"
      });
    }

    return ContentService.createTextOutput(JSON.stringify({status: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: error.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
